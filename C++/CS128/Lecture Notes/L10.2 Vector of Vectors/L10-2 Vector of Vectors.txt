    std::vector<std::vector<int>> is a vector of vectors
    Declaring a vector of vectors
        const unsigned int kStudentCount = 2;
        const unsigned int kAssignmentCount = 3;
        const int kStartingScore = 0;

        std::vector<std::vector<int>> gradebook(kStudentCount,
                                        std::vector<int>(kAssignmentCount, kStartingScore));
        std::vector<std::vector<int>> roster(kStudentCount);

    Visiting every element
